# nvinfer configuration for TrafficCamNet (DetectNet_v2, pruned ResNet18).
#
# Derived from the stock DeepStream reference
#   /opt/nvidia/deepstream/deepstream/samples/configs/deepstream-app/config_infer_primary.txt
# with three deliberate differences, all explained below.
#
# Paths here are resolved by nvinfer relative to THIS file's directory.

[property]
gpu-id=0

# --- Model -------------------------------------------------------------------
# The prebuilt FP16 engine from Milestone 4. `trafficcamnet_fp16.engine` is a
# symlink maintained by scripts/ds_common.sh, pointing at the versioned engine
# the current environment expects (name encodes TensorRT version and GPU). That
# keeps this committed file free of any version string.
model-engine-file=../models/engines/trafficcamnet_fp16.engine

# DIFFERENCE 1: `onnx-file` is deliberately NOT set.
# With no ONNX to fall back to, a failed engine load cannot silently rebuild --
# it fails loudly instead. This milestone never builds an engine.

# DIFFERENCE 2: `int8-calib-file` is deliberately NOT set.
# It applies only to an INT8 build. This is an FP16 engine.

# Read through the unversioned /opt/nvidia/deepstream/deepstream symlink, so no
# DeepStream version is hard-coded.
labelfile-path=/opt/nvidia/deepstream/deepstream/samples/models/Primary_Detector/labels.txt

# --- Preprocessing -----------------------------------------------------------
# 1/255: pixels are scaled to [0,1]. No mean subtraction -- TrafficCamNet was
# trained without it, unlike several other TAO models in the same install.
net-scale-factor=0.00392156862745098
# 0 = RGB
model-color-format=0

# --- Batching ----------------------------------------------------------------
# Must be 1. The engine was built with a fixed optimisation profile
# (min=opt=max=1x3x544x960), and nvinfer requires
# engine maxBatchSize >= configured batch-size.
batch-size=1

# --- Network -----------------------------------------------------------------
# 0 = primary (operate on full frames)
process-mode=1
# 0=FP32 1=INT8 2=FP16. Descriptive only when loading a prebuilt engine: the
# engine already fixes its precisions. Recorded so the config states its intent.
network-mode=2
# 0 = detector (default, stated explicitly)
network-type=0
num-detected-classes=4
# 0 = run inference on every frame
interval=0
gie-unique-id=1

# DIFFERENCE 3: `infer-dims` and `output-blob-names` are deliberately NOT set.
# nvinfer only validates input dimensions when infer-dims is present, and only
# checks output names when they are configured. Omitting both lets the engine's
# own bindings govern and removes two chances for a spurious mismatch.

# No parse-bbox-func-name and no custom-lib-path: DetectNet_v2's two-tensor
# output (output_cov/Sigmoid:0, output_bbox/BiasAdd:0) is decoded by nvinfer's
# BUILT-IN parser, which locates the box tensor by the substring "bbox" in its
# name. This is why TrafficCamNet needs no compiled custom parser.

# --- Clustering --------------------------------------------------------------
# Every grid cell overlapping an object fires, so raw output is a voting field
# rather than a detection list. Clustering is mandatory, not optional.
# 1=DBSCAN 2=NMS 3=DBSCAN+NMS 4=None
cluster-mode=2

# Thresholds are the stock reference values, unchanged. Milestone 5 records raw
# detection counts at these settings before considering any tuning.
[class-attrs-all]
topk=20
nms-iou-threshold=0.5
pre-cluster-threshold=0.2

# Stock also raises the threshold for class 0 (car) only. Kept for fidelity to
# the reference; it does not affect class 2 (person).
[class-attrs-0]
topk=20
nms-iou-threshold=0.5
pre-cluster-threshold=0.4
