# deepstream-app: SOAK pipeline, Milestone 8.2.
#
#   file source -> decoder -> nvstreammux -> nvinferserver (Triton)
#                -> nvtracker -> nvdsanalytics -> nvmultistreamtiler
#                -> nvdsosd -> fakesink
#
# This is deepstream_app_walk_triton.txt with EXACTLY THREE CHANGES. Nothing
# about the application under measurement is allowed to differ, because the
# point of Milestone 8.2 is to characterise the DEPLOYED pipeline, not a
# variant of it:
#
#   1. [tests] file-loop=1        the clip repeats instead of ending at EOS
#   2. the four KITTI *-output-dir keys are REMOVED
#   3. perf-measurement-interval-sec  5 -> 1
#
# Everything else -- source, streammux, inference config, tracker, analytics,
# ROI, tiler, OSD, sink, per-class colours -- is byte-for-byte the Milestone 7
# file. Diff it against deepstream_app_walk_triton.txt to confirm that claim.
#
# WHY each change:
#
#   file-loop=1     A 288-frame clip cannot exercise steady state. There is no
#                   CLI override for this key -- deepstream-app accepts only -c
#                   and -i -- so a looping run needs its own config file. The
#                   loop is implemented by DeepStream as a seek, not a source
#                   restart, and it is NOT gapless: one seek per 288 frames is
#                   part of what this run measures, and is accounted for by
#                   differencing against the single-pass Phase A figure.
#
#   no dumps        At the throughput this pipeline reaches, a ten-minute run
#                   is tens of thousands of frames. Writing one KITTI file per
#                   frame would cost hundreds of MB across hundreds of
#                   thousands of files, and -- worse -- would put synchronous
#                   per-frame disk I/O INSIDE the thing being measured.
#                   Correctness is not under test here; Milestone 7 proved it.
#
#   interval 1      One PERF sample per second rather than per five, so that
#                   throughput TREND over the run is resolvable and the
#                   file-loop artifact is visible in the distribution rather
#                   than smeared uniformly across every sample.
#
# This file is NOT baked into the container image. It is bind-mounted read-only
# at run time by scripts/measure_edge.sh, so no image is rebuilt.
#
# Paths are resolved relative to THIS file's directory.

[application]
enable-perf-measurement=1
perf-measurement-interval-sec=1

# A 1x1 tiler. Not needed to *tile* anything -- there is one source -- but
# required for correct OSD rendering. Without it, nvdsosd draws into a buffer it
# received by reference and previous frames' boxes persist as a visible trail.
# Full experiment and evidence: docs/milestone-05-osd-ghosting.md
[tiled-display]
enable=1
rows=1
columns=1
width=1920
height=1080
gpu-id=0
nvbuf-memory-type=0

# Type 2 = URI. The unversioned /opt/nvidia/deepstream/deepstream symlink keeps
# the DeepStream version out of this file.
[source0]
enable=1
type=2
uri=file:///opt/nvidia/deepstream/deepstream/samples/streams/sample_walk.mov
gpu-id=0
cudadec-memtype=0

# Type 1 = FakeSink. Genuinely consumes buffers, opens no window.
# sync=0: UNPACED. This run measures maximum sustainable throughput and is the
# harshest thermal and power condition -- it is not, and must not be described
# as, real-time playback.
[sink0]
enable=1
type=1
sync=0
source-id=0
gpu-id=0

# OSD stays enabled even headless: it is part of the deployed pipeline, so
