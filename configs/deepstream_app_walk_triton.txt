# deepstream-app: HEADLESS pipeline with IN-PROCESS TRITON, Milestone 7.
#
#   file source -> decoder -> nvstreammux -> nvinferserver (Triton)
#                -> nvtracker -> nvdsanalytics -> nvmultistreamtiler
#                -> nvdsosd -> fakesink
#
# This is deepstream_app_walk_headless.txt with EXACTLY TWO LINES CHANGED, in
# [primary-gie]: plugin-type=1 and the config-file. Everything else -- source,
# streammux, tracker, analytics, ROI, tiler, OSD, sink, dump directories,
# per-class colours -- is byte-for-byte the Milestone 6 file, because Milestone
# 7 replaces only the inference-serving layer.
#
# Diff it against deepstream_app_walk_headless.txt to confirm that claim.
#
# No analytics, no secondary inference. Bounded: one pass of a 9.61 s clip to
# natural EOS, never looping.
#
# Paths are resolved relative to THIS file's directory.

[application]
enable-perf-measurement=1
perf-measurement-interval-sec=5

# Four KITTI metadata dumps, written by FOUR DIFFERENT probes at different
# points in the pipeline. That is what makes it possible to compare detector
# output against tracker output from a SINGLE run.
#
# One file per frame in each directory: <appIdx>_<streamId>_<frameNum>.txt
# The directories must already exist; the verification scripts create them.

# DETECTOR output. Written from a probe on the PRIMARY-GIE bin's src pad
# (deepstream_app.c:1951), i.e. after nvinfer and strictly before the tracker,
# so adding a tracker cannot change what lands here. No object_id column.
#   label 0.0 0 0.0 left top right bottom 0.0 x6 confidence
gie-kitti-output-dir=../models/detections

# TRACKER output. Written from a probe on the src pad of the last common
# element (deepstream_app.c:1979) -- the tracker, once one exists. Same format
# plus the tracking id in column 2, the tracker's box, and tracker confidence.
#   label id 0.0 0 0.0 left top right bottom 0.0 x7 confidence
kitti-track-output-dir=../models/tracks

# Tracks the low-level library reports as terminated, and frames it reports as
# shadow-tracked (a target kept alive while the detector did not see it).
# Both are evidence for behaviour that CANNOT be assumed; see
# docs/milestone-05-tracking.md.
terminated-track-output-dir=../models/tracks_terminated
shadow-track-output-dir=../models/tracks_shadow

# A 1x1 tiler. Not needed to *tile* anything -- there is one source -- but
# required for correct OSD rendering. Without it, nvdsosd draws into a buffer it
# received by reference (osd_conv negotiates NV12 -> NV12 and converts nothing),
# and previous frames' boxes persist as a visible trail. The tiler composites
# into a fresh buffer, which removes the artifact while keeping the GPU draw
# path. Every stock NVIDIA rendering config uses a tiler for the same reason.
# Full experiment and evidence: docs/milestone-05-osd-ghosting.md
[tiled-display]
enable=1
rows=1
columns=1
width=1920
height=1080
gpu-id=0
nvbuf-memory-type=0

# Type 2 = URI. The unversioned /opt/nvidia/deepstream/deepstream symlink keeps
# the DeepStream version out of this file.
[source0]
enable=1
type=2
uri=file:///opt/nvidia/deepstream/deepstream/samples/streams/sample_walk.mov
gpu-id=0
cudadec-memtype=0

# Type 1 = FakeSink. Genuinely consumes buffers, opens no window.
# sync=0: verification should not wait out the clip's 9.61 s in real time.
[sink0]
enable=1
type=1
sync=0
source-id=0
gpu-id=0

# OSD stays enabled even headless. It is what makes nvinfer's metadata be drawn,
# and the KITTI probe attaches to its sink pad.
[osd]
enable=1

text-size=15

text-bg-color=0.3;0.3;0.3;1

# width/height are the MUX OUTPUT resolution, i.e. the source resolution --
# NOT the 960x544 network input. nvinfer scales to the network size itself.
# live-source=0 because this is a file, not a live camera.
[streammux]

[primary-gie]

# plugin-type=1 selects nvinferserver instead of nvinfer. This single key is the

# links libtritonserver.so and runs Triton IN-PROCESS via its C API.
plugin-type=1
config-file=config_inferserver_trafficcamnet.txt
# Per-class box colours. Class 2 = person, drawn blue.
bbox-border-color0=1;0;0;1

# Milestone 5 checkpoint 2. deepstream-app places the tracker immediately after
# the primary GIE and before the tiler; that ordering is built into
# create_common_elements() and is not configurable:
#
#   nvstreammux -> nvinfer -> nvtracker -> nvmultistreamtiler -> nvdsosd -> sink
#
# DeepStream 9.1 ships ONE low-level tracker library for every backend; the
# backend is chosen entirely by the YAML given to ll-config-file. Both are
# NVIDIA's own files, referenced through the unversioned /opt symlink and used
# exactly as installed -- neither is copied into this repository nor modified.
# scripts/ds_common.sh preflight-checks that both exist.
#
# NvSORT = Kalman motion model + IoU/size association, no appearance model and
# no ReID network, so nothing has to be downloaded or built. Why NvSORT rather
# than IOU or NvDCF: docs/milestone-05-tracking.md
#
# NOTE: enable-batch-process and enable-past-frame DO NOT EXIST in DeepStream
# 9.1 -- they are absent from both the element and the config parser, and would
# be silently ignored. Batching is unconditional; past-frame output is decided
# by the low-level library's capability query.
[tracker]
enable=1
# The tracker's own working resolution. It matches neither the source nor the
# network by requirement: bbox coordinates in metadata stay in streammux
# (1920x1080) coordinates. 960x544 is the element default and NVIDIA's sample
# value; NvDCF/NvDeepSORT additionally require multiples of 32.
tracker-width=960
tracker-height=544
ll-lib-file=/opt/nvidia/deepstream/deepstream/lib/libnvds_nvmultiobjecttracker.so
ll-config-file=/opt/nvidia/deepstream/deepstream/samples/configs/deepstream-app/config_tracker_NvSORT.yml
gpu-id=0
display-tracking-id=1

# Milestone 5 checkpoint 3. deepstream-app places nvdsanalytics immediately
# after the tracker and before the tiler; like the tracker's position, that
# ordering is built into create_common_elements() and is not configurable:
#
#   nvinfer -> nvtracker -> nvdsanalytics -> nvmultistreamtiler -> nvdsosd
#
# Analytics only ANNOTATES. It attaches NvDsAnalyticsObjInfo per object and
# NvDsAnalyticsFrameMeta per frame, and never removes objects from the metadata
# -- despite what NVIDIA's own sample config comment claims.
#
# deepstream-app can RUN this but cannot report it: it never reads
# NVDS_USER_OBJ_META_NVDSANALYTICS or NVDS_USER_FRAME_META_NVDSANALYTICS, so the
# ROI verdict appears in no dump and no log. tools/analytics_probe.cpp exists
# solely to read it back. See docs/milestone-05-restricted-zone.md
[nvds-analytics]
enable=1
config-file=config_nvdsanalytics_restricted_zone.txt

# Never loop: verification must terminate on its own.
[tests]
file-loop=0
